// ---------------- USERS COLLECTION ----------------
db.createCollection("users", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["name", "email", "age", "address", "isActive"],
      properties: {
        name: { bsonType: "string" },
        email: { bsonType: "string" },
        age: { bsonType: "int", minimum: 18, maximum: 60 },
        isActive: { bsonType: "bool" },
        address: {
          bsonType: "object",
          required: ["city", "pincode"],
          properties: {
            city: { bsonType: "string" },
            pincode: { bsonType: "int" }
          }
        }
      }
    }
  }
})

// ---------------- ORDERS COLLECTION ----------------
db.createCollection("orders", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["user_id", "products", "total_amount", "status", "created_at"],
      properties: {
        user_id: { bsonType: "objectId" },
        total_amount: { bsonType: "number", minimum: 0 },
        status: {
          enum: ["PLACED", "CANCELLED"]
        },
        created_at: { bsonType: "date" },
        products: {
          bsonType: "array",
          minItems: 1,
          items: {
            bsonType: "object",
            required: ["name", "price", "quantity"],
            properties: {
              name: { bsonType: "string" },
              price: { bsonType: "number", minimum: 0 },
              quantity: { bsonType: "int", minimum: 1 }
            }
          }
        }
      }
    }
  }
})

// =====================================
// 8.1 WHY AGGREGATION PIPELINE IS POWERFUL
// =====================================

// Concept:
// Aggregation pipeline processes data in stages (like a pipeline)

// Example:
print("---- Pipeline Example ----")
db.orders.aggregate([
  { $match: { status: "PLACED" } },     // Filter early
  { $group: { _id: "$user_id", total: { $sum: "$total_amount" } } },
  { $sort: { total: -1 } }
])

// WHY powerful?
// ✔ Single query instead of multiple queries
// ✔ Efficient for analytics
// ✔ Works like data processing pipeline



// =====================================
// 8.2 WHEN TO USE $unwind
// =====================================

// Use $unwind when working with arrays

print("---- Without Unwind (Wrong for product analytics) ----")
db.orders.aggregate([
  {
    $group: {
      _id: "$products.name", // ❌ Won’t work properly (array)
      count: { $sum: 1 }
    }
  }
])

print("---- With Unwind (Correct) ----")
db.orders.aggregate([
  { $unwind: "$products" }, // ✔ flatten array
  {
    $group: {
      _id: "$products.name",
      total_qty: { $sum: "$products.quantity" }
    }
  }
])

// RULE:
// Use $unwind when:
// ✔ Field is array
// ✔ You need per-item analysis



// =====================================
// 8.3 SQL GROUP BY vs MongoDB $group
// =====================================

// SQL:
// SELECT user_id, SUM(total_amount)
// FROM orders
// GROUP BY user_id;

// MongoDB:
print("---- MongoDB GROUP ----")
db.orders.aggregate([
  {
    $group: {
      _id: "$user_id",
      total_spent: { $sum: "$total_amount" }
    }
  }
])

// Mapping:
// SQL GROUP BY → Mongo $group
// SQL SUM() → Mongo $sum
// SQL COUNT() → Mongo $sum:1



// =====================================
// 8.4 PERFORMANCE OPTIMIZATION
// =====================================

// 🔥 IMPORTANT RULES

// 1. USE $match EARLY
print("---- Optimized Query (match early) ----")
db.orders.aggregate([
  { $match: { status: "PLACED" } }, // ✔ filter first
  {
    $group: {
      _id: "$user_id",
      total: { $sum: "$total_amount" }
    }
  }
])

// BAD (slow):
print("---- Non-Optimized Query ----")
db.orders.aggregate([
  {
    $group: {
      _id: "$user_id",
      total: { $sum: "$total_amount" }
    }
  },
  { $match: { total: { $gt: 10000 } } }
])

// WHY?
// Filtering early reduces data processed



// 2. USE INDEXES
print("---- Create Index ----")
db.orders.createIndex({ user_id: 1 })

// Test performance
print("---- Explain Query ----")
db.orders.explain("executionStats").aggregate([
  { $match: { user_id: ObjectId("REPLACE_ID") } }
])


// 3. AVOID UNNECESSARY $unwind
// $unwind increases number of documents → can slow queries



// 4. USE $project TO REDUCE DATA
print("---- Use Project Early ----")
db.orders.aggregate([
  {
    $project: {
      user_id: 1,
      total_amount: 1
    }
  },
  {
    $group: {
      _id: "$user_id",
      total: { $sum: "$total_amount" }
    }
  }
])


// 5. LIMIT DATA WHEN POSSIBLE
print("---- Limit Example ----")
db.orders.aggregate([
  { $sort: { total_amount: -1 } },
  { $limit: 5 }
])


// =====================================
// FINAL SUMMARY (IMPORTANT)
// =====================================

// ✔ Use $match early
// ✔ Use indexes
// ✔ Use $unwind only when needed
// ✔ Use $project to reduce data
// ✔ Use $limit for large datasets
