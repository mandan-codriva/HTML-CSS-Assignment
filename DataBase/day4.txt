create table users(
user_id int primary key auto_increment,
name varchar(30) not null ,
email varchar(30) unique not null,
phone varchar(15) unique not null,
created_at timestamp default current_timestamp,
updated_at timestamp default current_timestamp
);

create table products(
product_id int primary key auto_increment,
name varchar(50) not null,
price decimal(10,2) not null check(price>0),
stock int not null check(stock>=0),
created_at timestamp default current_timestamp,
updated_at  timestamp default current_timestamp on update current_timestamp
);

create table carts(
cart_id int primary key auto_increment,
user_id int not null unique,
created_at timestamp default current_timestamp,
updated_at  timestamp default current_timestamp on update current_timestamp,
foreign key (user_id) references users(user_id) on delete cascade
);

create table cartItems(
cartItem_id int primary key auto_increment,
cart_id int not null,
product_id int not null,
quantity int not null check(quantity>0),
price decimal(10,2) check(price>0) not null,
foreign key (cart_id) references carts(cart_id) on delete cascade,
foreign key (product_id) references products(product_id), 
unique(cart_id,product_id),
created_at timestamp default current_timestamp,
updated_at  timestamp default current_timestamp on update current_timestamp
);



create table orders (
  order_id int primary key auto_increment,
  user_id int not null,
  order_date date not null,
  status varchar(30) not null 
  check (status in ('PENDING', 'COMPLETED', 'CANCELLED')),
  total_amount decimal(10,2) not null check (total_amount >= 0),
  created_at timestamp default current_timestamp,
  updated_at timestamp default current_timestamp on update current_timestamp,
  foreign key (user_id) references users(user_id)
);

create table orderItem(
orderItem_id int primary key auto_increment,
order_id int not null,
product_id int not null,
unique(order_id,product_id),
quantity int check(quantity>0) not null,
price decimal(10,2) check (price>0) not null,
foreign key (order_id) references orders(order_id) on delete cascade,
foreign key (product_id) references products(product_id),
created_at timestamp default current_timestamp,
updated_at  timestamp default current_timestamp on update current_timestamp
);

-- Deep understanding of grouping logic --

-- Take a messy table and organize rows into buckets, then do calculations on each bucket. 
-- GROUP BY transforms a set of rows into a smaller set of summarized rows by:
-- Partitioning the data into groups based on one or more columns
-- Performing aggregation within each group
-- Returning one result row per group
-- GROUP BY means “divide data into groups and calculate something for each group.
-- Without GROUP BY → results are calculated over the entire dataset
-- With GROUP BY → results are calculated per group
select * from orders;

-- Difference between WHERE vs HAVING

-- WHERE Clause

-- The WHERE clause is used to filter rows before grouping and aggregation take place. It operates on individual rows and determines which rows should be included in the result set prior to any grouping.

-- Works on row-level data
-- Cannot use aggregate functions (SUM, COUNT, AVG, etc.)
-- Applied before GROUP BY

--  HAVING Clause

-- The HAVING clause is used to filter groups after grouping and aggregation have been performed. It operates on aggregated data and determines which groups should be included in the final result.

-- Works on group-level data
-- Can use aggregate functions
-- Applied after GROUP BY

-- Using Aggregate Functions Without GROUP BY

-- Using aggregate functions with non-aggregated columns without GROUP BY
-- select user_id, sum(total_amount)
-- from orders;
-- Why this is wrong
-- SUM(total_amount) → gives one value
-- user_id → has multiple values
-- SQL gets confused:
-- “Should I return one row or many rows?”

-- 2. Wrong Grouping Column

-- Wrong Grouping Column
-- Problem

-- Grouping by the wrong column changes the meaning
-- select user_id, sum(total_amount)
-- from orders
-- group by order_id;
-- Why this is wrong
-- order_id is unique
-- So each group = one row

-- 👉 Result:

-- SUM(total_amount) = same as original value
-- No real aggregation happens 
-- Fix
-- group by user_id;

-- Now grouping matches the requirement:
-- “Total per user”

--  3.

-- All aggregation errors come from one root issue:
-- Mixing row-level data with group-level data
-- Row level	Individual records
-- Group level	Aggregated summary
-- “Group data first → then calculate → then display one result per group”

-- Aggregation errors occur when SQL cannot reconcile multiple row-level values into a single group-level result due to improper grouping or misuse of aggregate functions.

-- COUNT(*) vs COUNT(column)
-- COUNT(*)
-- Counts all rows, including rows where columns contain NULL
-- Does not depend on any specific column

-- Definition:
-- COUNT(*) returns the total number of rows in a dataset.

-- COUNT(column)
-- Counts only non-NULL values in that column
-- Ignores rows where the column value is NULL
--  Definition:
-- COUNT(column) returns the number of rows where the specified column is not NULL.

-- COUNT(*) measures dataset size, whereas COUNT(column) measures data completeness for a specific attribute.

-- 2. Importance of GROUP BY
--  Definition

-- GROUP BY is used to organize data into groups based on one or more columns, allowing aggregate functions to be applied to each group.
--  Why it is needed
-- Without GROUP BY:
-- Aggregate functions operate on the entire dataset
-- With GROUP BY:
-- Aggregation happens per group
-- Conceptual Insight
-- GROUP BY changes the level of analysis from row-level to group-level, enabling meaningful summaries such as totals, averages, and counts for each category.


-- HAVING vs WHERE
--   WHERE
-- Filters rows before grouping
-- Cannot use aggregate functions
-- Definition:
-- WHERE restricts the dataset prior to aggregation.
-- HAVING
-- Filters groups after aggregation
-- Can use aggregate functions
--  Definition:
-- HAVING filters grouped results based on aggregate conditions.

-- 4. Impact of Incorrect GROUP BY
-- Problem
-- Using wrong grouping columns leads to:
-- Incorrect aggregation
-- Misleading results
-- Loss of analytical meaning
-- Common mistakes
-- 1. Over-grouping
-- Grouping by too many columns:
-- Breaks data into very small groups
-- Reduces usefulness of aggregation
-- 2. Under-grouping
-- Grouping by too few columns:
-- Merges unrelated data
-- Produces incorrect summaries
-- 3. Grouping by unique column
-- Each group has only one row
-- Aggregation becomes meaningless
-- Conceptual Insight
-- The choice of GROUP BY columns determines the granularity of analysis; incorrect grouping leads to either over-fragmented or overly generalized results.
