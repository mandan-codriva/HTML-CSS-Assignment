create database transactions;

CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL
);


CREATE TABLE accounts (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id),
    balance DECIMAL(12,2) CHECK (balance >= 0)
);

CREATE TABLE transactions (
    id SERIAL PRIMARY KEY,
    from_account_id INT REFERENCES accounts(id),
    to_account_id INT REFERENCES accounts(id),
    amount DECIMAL(12,2) CHECK (amount > 0),
    status VARCHAR(20),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Transaction A
BEGIN;

-- Lock sender account
SELECT * FROM accounts
WHERE id = 1
FOR UPDATE;

UPDATE accounts
SET balance = balance - 3000
WHERE id = 1
  AND balance >= 3000;

UPDATE accounts
SET balance = balance + 3000
WHERE id = 2;

COMMIT;



-- Transaction B (runs simultaneously)
BEGIN;

-- Must wait until Transaction A completes
SELECT * FROM accounts
WHERE id = 1
FOR UPDATE;

UPDATE accounts
SET balance = balance - 2500
WHERE id = 1
  AND balance >= 2500;

UPDATE accounts
SET balance = balance + 2500
WHERE id = 3;

COMMIT;



-- Without FOR UPDATE:
-- Both transactions may read same balance
-- Overspending possible

-- With FOR UPDATE:
-- First transaction locks row
-- Second waits
-- Prevents race conditions
-- Ensures safe balance usage
