CREATE DATABASE ecommerce_indexing;

CREATE TABLE users (
    user_id SERIAL PRIMARY KEY,
    full_name VARCHAR(100),
    email VARCHAR(150) UNIQUE,
    is_active BOOLEAN DEFAULT true,
    preferences JSONB,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE products (
    product_id SERIAL PRIMARY KEY,
    product_name VARCHAR(150),
    category VARCHAR(100),
    price NUMERIC(10,2),
    stock_quantity INT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

create table orders(
 order_id serial primary key,
 user_id int references users(user_id),
 order_date timestamp default current_timestamp,
 status varchar(50),
 total_amount numeric(10,2)
);

CREATE TABLE order_items (
    order_item_id SERIAL PRIMARY KEY,
    order_id INT REFERENCES orders(order_id),
    product_id INT REFERENCES products(product_id),
    quantity INT,
    item_price NUMERIC(10,2)
);

--What Is an Index?
--
--A PostgreSQL index is a special data structure that improves data retrieval speed.
--
--Without Index:
--
--PostgreSQL scans every row:
--
--Seq Scan
--With Index:
--
--PostgreSQL directly locates rows:
--
--Index Scan
--Real-Life Analogy
--
--A book index:
--
--Without index → read every page
--With index → jump directly to topic page
--2. Why Indexes Improve Performance
--
--Indexes reduce:
--
--Full table scans
--Disk reads
--CPU workload
--Query response time
--Best For:
--WHERE
--JOIN
--ORDER BY
--GROUP BY
--Example:
--SELECT * FROM users WHERE email = 'user5000@gmail.com';
--
--Indexed email makes this much faster.
--
--3. When Indexes Are Not Useful
--
--Indexes may not help when:
--
--Small tables:
--
--Sequential scans may be faster.
--
--Low-selectivity columns:
--
--Example:
--
--is_active = true
--
--If most rows match, index adds little value.
--
--Heavy write systems:
--
--Too many inserts/updates make index maintenance expensive.
--
--Functions in queries:
--LOWER(email)
--4. Types of Indexes
--Single-Column Index
--CREATE INDEX idx_users_email ON users(email);
--Good for:
--WHERE email = ?
--Composite Index
--CREATE INDEX idx_orders_user_status ON orders(user_id, status);
--Good for:
--WHERE user_id = ? AND status = ?
--Important:
--
--Column order matters.
--
--Functional Index
--CREATE INDEX idx_lower_email ON users(LOWER(email));
--Partial Index
--CREATE INDEX idx_active_users
--ON users(email)
--WHERE is_active = true;
--JSONB Index
--
--Useful for structured JSON fields.
--
--5. Trade-Offs of Indexing
--Benefit	Drawback
--Faster SELECT	Slower INSERT
--Faster filtering	Slower UPDATE
--Better JOINs	More storage
--Better analytics	Maintenance overhead
--Over-Indexing Problems
--
--Too many indexes cause:
--
--Slower writes
--Higher storage usage
--More vacuum/reindex cost
--Planner confusion
--6. Common PostgreSQL Scan Types
--Seq Scan
--
--Reads full table
--
--Index Scan
--
--Direct lookup
--
--Bitmap Index Scan
--
--Efficient for multiple row matches
--
--Index Only Scan
--
--Fastest when table access unnecessary
--
--7. Core Optimization Rule
--Index:
--Frequently searched columns
--Foreign keys
--JOIN keys
--Large tables
--High-selectivity columns
--Avoid:
--Rarely queried columns
--Frequently updated columns
--Very small tables
--Over-indexing
--8. Performance Formula
--
--In general:
--
--Better Query Speed≈Good Index Design+Efficient Query Structure

CREATE MATERIALIZED VIEW user_order_summary AS
SELECT
    u.user_id,
    u.full_name,
    COUNT(o.order_id) AS total_orders,
    SUM(o.total_amount) AS total_revenue
FROM users u
JOIN orders o ON u.user_id = o.user_id
GROUP BY u.user_id, u.full_name;


SELECT *
FROM user_order_summary
ORDER BY total_revenue DESC;
