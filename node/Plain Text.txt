 What is JavaScript?

JavaScript is a programming language mainly used to make websites interactive and dynamic. It runs inside web browsers and can also run outside browsers using environments like Node.js.
 With JavaScript, developers can create features such as buttons, forms, animations, games, APIs, and real-time applications.

Real-Life Analogy

HTML = Skeleton (structure)
CSS = Clothes and appearance (design)
JavaScript = Brain and muscles (behavior and actions)

Without JavaScript, a webpage would just sit there like a statue. JavaScript gives it life and interaction.

Example:

Clicking a button
Showing a popup
Updating content without refreshing
Playing videos automatically

For Array Specific

Recurssive approach


function arrayRecurssiveFlatern(input){

    const flattern = [];

    if(Array.isArray(input)){

        input.forEach(item => {

            const output = arrayRecurssiveFlatern(item);

            flattern.push(...output);

        });

    } else {

        flattern.push(input);

    }

    return flattern;
}



Iterative approach:

function flattenArrayIteratively(arr) {
  const stack = [...arr];
  const flattened = [];

  while (stack.length) {
    const current = stack.shift();

    if (Array.isArray(current)) {
      stack.unshift(...current);
    } else {
      flattened.push(current);
    }
  }

  return flattened;
}





function deepFlatten(obj) {
  const result = {};

  function helper(current, path) {

    // Handle null
    if (current === null) {
      result[path] = null;
      return;
    }

    // Handle arrays
    if (Array.isArray(current)) {

      current.forEach((item, index) => {

        helper(item, `${path}[${index}]`);

      });

      return;
    }

    // Handle objects
    if (typeof current === "object") {

      for (const key in current) {

        const newPath = path
          ? `${path}.${key}`
          : key;

        helper(current[key], newPath);
      }

      return;
    }

    // Transform strings
    if (typeof current === "string") {
      result[path] = current.toUpperCase();
    } else {
      result[path] = current;
    }
  }

  helper(obj, "");

  return result;
}
