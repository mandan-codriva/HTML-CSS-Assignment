14. Create a Simple HTTP Server
Code

    response.writeHead(200, {
        "Content-Type": "text/plain"
    });

    response.end("Hello Node.js Core");
});

server.listen(3000, () => {
    console.log("Server is running on port 3000");
});

19. Export Multiple Functions using Destructuring
multipleExports.js
function add(a = 0, b = 0) {
    return a + b;
}

function multiply(a = 0, b = 0) {
    return a * b;
}

function divide(a = 0, b = 1) {
    return a / b;
}

module.exports = {
    add,
    multiply,
    divide
};
index.js
const { add, multiply, divide } = require('./multipleExports');

console.log("Addition:", add(5, 3));
console.log("Multiplication:", multiply(5, 3));
console.log("Division:", divide(10, 2));
